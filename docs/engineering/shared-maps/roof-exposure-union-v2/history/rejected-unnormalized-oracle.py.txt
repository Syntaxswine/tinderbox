"""Prove optimized operations and old successful source masks before any rear attempt."""
from support import *
def synthetic():
    G=module("unmodified_polygon_controls",FIRST/"geometry.py")
    G.deadline=time.monotonic()+55;engine=ExactUnion(G)
    old_intersection=engine.base_intersection;old_union=G.union_cells
    def rect(x,z,w,d):return [(F(x),F(z)),(F(x+w),F(z)),(F(x+w),F(z+d)),(F(x),F(z+d))]
    unit=rect(0,0,1,1);tiny=F(1,1<<80)
    cases={
      "cyclic-reversed-duplicate":[unit,unit[1:]+unit[:1],list(reversed(unit))],
      "strict-contained":[rect(-1,-1,3,3),unit,rect(F(1,4),F(1,4),F(1,4),F(1,4))],
      "boundary-touch":[unit,rect(1,0,1,1)],
      "positive-tiny-outside-strip":[unit,rect(1-tiny,0,2*tiny,1)],
      "positive-tiny-island":[unit,rect(2,2,tiny,tiny)],
      "crossed-partial-overlap":[unit,rect(F(1,2),F(1,2),1,1)],
      "bbox-is-not-containment":[[(F(0),F(0)),(F(2),F(0)),(F(0),F(2))],rect(F(3,4),F(3,4),1,1)]
    }
    baseline={}
    for name,polys in cases.items():
        records=[{"owner":str(i),"sourceOffset":i,"polygon":G.normalized(p)} for i,p in enumerate(polys)]
        baseline[name]=old_union(records)
    half_identity=G.half;engine.install();outcomes=[]
    for name,polys in cases.items():
        records=[{"owner":str(i),"sourceOffset":i,"polygon":G.normalized(p)} for i,p in enumerate(polys)]
        before=G.clips;expected=baseline[name];actual=engine.union_cells(records)
        equivalent(G,old_intersection,[r["polygon"] for r in expected],[r["polygon"] for r in actual])
        for i,a in enumerate(polys):
            for b in polys[:i]:
                plain_intersection=old_intersection(a,b);fast_intersection=engine.intersection(a,b)
                equivalent(G,old_intersection,[plain_intersection] if G.area(plain_intersection) else [],[fast_intersection] if G.area(fast_intersection) else [])
                equivalent(G,old_intersection,engine.base_difference(a,b),engine.difference(a,b))
        area=sum((G.area(r["polygon"]) for r in actual),F(0))
        if name=="positive-tiny-outside-strip":assert area==1+tiny
        if name=="positive-tiny-island":assert area==1+tiny*tiny
        outcomes.append({"case":name,"setEquivalent":True,"area":area,"clipsIncludingReference":G.clips-before})
    # Containment shortcuts must reject the bbox-only trap and preserve strict tiny height.
    assert not engine.contains(cases["bbox-is-not-containment"][0],cases["bbox-is-not-containment"][1])
    assert G.half(unit,(F(0),F(0),tiny))==unit and G.area(G.half(unit,(F(0),F(0),-tiny)))==0
    assert G.half is half_identity
    clip_probe=G.clips;G.half(unit,(F(0),F(0),F(1)));assert G.clips==clip_probe+1
    save("synthetic-checks.json",{"passed":True,"cases":outcomes,"positiveTinyHeightRetained":True,
                                 "allClipsIncludingReference":G.clips,"optimizer":engine.snapshot()})
    print(json.dumps({"syntheticChecksPassed":True,"cases":len(cases),"clipsIncludingReference":G.clips}))
def source(slug):
    assert slug in ("end-0","end-1")
    inputs=pins();E,engine,captured=run(slug);G=E.G
    new=captured[slug+"/partitions.json"]["partitions"];old=old_partitions(slug)
    for r in new:
        c=[p["polygon"] for p in r["coverCells"]];e=r["exposedCells"]
        equivalent(G,engine.base_intersection,c,old[r["roofOffset"]][0])
        equivalent(G,engine.base_intersection,e,old[r["roofOffset"]][1])
    result=captured[slug+"/result.json"];prior=read(OLD/(slug+"/result.json"))
    assert result["areaScalarDirectedLower"]["bits"]==prior["areaScalarDirectedLower"]["bits"]
    assert time.monotonic()<G.deadline,"Unchanged deadline"
    check={"passed":True,"target":slug,"sourceInputs":inputs,"sameSixSourceRoofFaces":True,"coverAndComplementSetsEquivalent":True,
           "areaScalarBits":result["areaScalarDirectedLower"]["bits"],"clipsIncludingEquivalence":G.clips,
           "optimizer":engine.snapshot()}
    save(slug+"/preflight.json",check)
    print(json.dumps({"sourcePreflightPassed":True,"target":slug,"clipsIncludingEquivalence":G.clips,
                      "helpers":engine.snapshot()["helperCounts"],"scalarBits":check["areaScalarBits"]}))
if __name__=="__main__":
    if sys.argv[1]=="synthetic":synthetic()
    else:source(sys.argv[1])
