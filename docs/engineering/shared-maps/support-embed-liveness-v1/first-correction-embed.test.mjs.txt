import test from 'node:test';
import assert from 'node:assert/strict';
import {createOriginalSupportPorts} from '../../src/simulation/original-support-ports.js';
import {boxTriangles} from '../fixtures/regional-query-data.mjs';
import {clipTriangle,bounds} from '../../workbenches/furnished-house/geometry.js';
import {supportSurfacesTouch} from '../../workbenches/furnished-house/support.js';
import {sweptCylinderClear} from '../../src/runtime/collision.js';

const point=[1,.5,.5];
const bottom=(p,y)=>p.triangles.flatMap((t,i)=>t.every(v=>v[1]===y)?[i]:[]);
const shape=ts=>({triangles:ts,solidTriangles:ts,bounds:bounds(ts.flat()),normal:[0,1,0],kind:'wall',thickness:0});
const inside=(p,ts)=>!sweptCylinderClear({x:p[0],y:p[2]},{x:p[0],y:p[2]},0,p[1],p[1],{triangles:ts,closed:true});
function fixture(mode='cold'){
  const parent={id:'parent',ownerId:'parent',triangles:boxTriangles([0,0,0],[2,1,1])},child={id:'child',ownerId:'child',triangles:boxTriangles([.8,.2,.2],[1.2,.8,.8])};
  const fragment=p=>({id:mode+'/'+p.id,recordId:'native/'+p.id,partId:p.id,ownerId:p.ownerId,component:0,mode,primitives:p.triangles.map((_,i)=>i),triangles:p.triangles});
  const endpoint=(p,y)=>({partId:p.id,point,primitives:bottom(p,y),nativeRecordId:'native/'+p.id});
  const core=createOriginalSupportPorts({parts:[parent,child],bonds:[{id:'embed',kind:'fixed-embed',direction:'parent-to-child',parent:endpoint(parent,0),child:endpoint(child,.2)}],anchors:[{partId:'parent',elevation:0}]});
  const clip=(p,min,max)=>{const ts=p.triangles.map(t=>clipTriangle(t,{min,max}));return {...fragment(p),triangles:ts.flat(),primitives:ts.flatMap((ts,i)=>ts.length?[i]:[])};};
  return {core,parent,child,fragment,clip};
}

test('architect repro: surviving original ports cannot bridge the removed strict embed',()=>{
  for(const mode of ['cold','native']){const {core,parent,child,fragment,clip}=fixture(mode);try{
    const full=[fragment(parent),fragment(child)],initial=core.evaluate({revision:0,fragments:full});assert.deepEqual(initial.unrootedIds,[]);assert.equal(initial.ports[0].live,true);
    const clipped=clip(parent,[0,0,0],[2,.1,1]);assert.equal(supportSurfacesTouch(shape(clipped.triangles),shape(child.triangles)),false);
    const result=core.evaluate({revision:1,fragments:[clipped,fragment(child)],bornRecordIds:['native/child']});
    assert.deepEqual(result.supportedIds,[mode+'/parent']);assert.deepEqual(result.unrootedIds,[mode+'/child']);assert.equal(result.ports[0].live,false);
    // A new call must use its own material, without retaining a prior verdict.
    assert.deepEqual(core.evaluate({revision:2,fragments:full}).unrootedIds,[]);
    if(mode==='native')assert.equal(result.births[0].state,'unrooted');
  }finally{core.dispose();}}
});

test('losing child material at the common witness also breaks the embed',()=>{
  for(const mode of ['cold','native']){const {core,parent,child,fragment,clip}=fixture(mode);try{
    const result=core.evaluate({revision:1,fragments:[fragment(parent),clip(child,[.8,.2,.2],[1.2,.3,.8])]});
    assert.deepEqual(result.unrootedIds,[mode+'/child']);assert.equal(result.ports[0].live,false);
  }finally{core.dispose();}}
});

test('open exterior facets fail closed without current closed material authority',()=>{
  const {core,parent,child,fragment}=fixture();try{
    const onlyBottom=p=>({...fragment(p),triangles:bottom(p,p===parent?0:.2).map(i=>p.triangles[i])});
    const result=core.evaluate({revision:0,fragments:[onlyBottom(parent),onlyBottom(child)]});
    assert.deepEqual(result.unrootedIds,['cold/child']);assert.equal(result.ports[0].live,false);
    assert.ok(result.unresolvedBindings.some(b=>b.reason==='strict-embed-effective-solid-unproven'));
  }finally{core.dispose();}
});

test('authoritative closed material can prove an embed for an exterior-only face view',()=>{
  for(const mode of ['cold','native']){const {core,parent,child,fragment}=fixture(mode);try{
    const fragmentWithMaterial=p=>({...fragment(p),triangles:bottom(p,p===parent?0:.2).map(i=>p.triangles[i]),solidTriangles:p.triangles});
    const result=core.evaluate({revision:0,fragments:[fragmentWithMaterial(parent),fragmentWithMaterial(child)]});
    assert.deepEqual(result.unrootedIds,[]);assert.equal(result.ports[0].live,true);
  }finally{core.dispose();}}
});

test('a real closed void at the original witness defeats unchanged exterior ports',()=>{
  const {core,parent,child,fragment}=fixture('native');try{
    // Explicit closed physical test input, not faces closed or generated by the core.
    const voidShell=boxTriangles([.9,.4,.4],[1.1,.6,.6]).map(t=>[...t].reverse()),solidTriangles=[...parent.triangles,...voidShell];
    assert.equal(inside(point,solidTriangles),false);
    const result=core.evaluate({revision:1,fragments:[{...fragment(parent),solidTriangles},fragment(child)]});
    assert.deepEqual(result.unrootedIds,['native/child']);assert.equal(result.ports[0].live,false);
    assert.ok(result.unresolvedBindings.some(b=>b.reason==='strict-embed-witness-not-in-surviving-interior'));
  }finally{core.dispose();}
});

test('surviving witness material cannot jump a void on its fixed original path',()=>{
  const {core,parent,child,fragment}=fixture('native');try{
    const voidShell=boxTriangles([.9,.1,.4],[1.1,.3,.6]).map(t=>[...t].reverse()),solidTriangles=[...parent.triangles,...voidShell];
    assert.equal(inside(point,solidTriangles),true);
    const result=core.evaluate({revision:1,fragments:[{...fragment(parent),solidTriangles},fragment(child)]});
    assert.deepEqual(result.unrootedIds,['native/child']);assert.equal(result.ports[0].live,false);
    assert.ok(result.unresolvedBindings.some(b=>b.reason==='strict-embed-original-path-cut'));
  }finally{core.dispose();}
});

test('different current closed components cannot share one original embed path',()=>{
  const {core,parent,child,fragment}=fixture('native');try{
    const solidTriangles=[...boxTriangles([0,0,0],[2,.1,1]),...boxTriangles([0,.4,0],[2,1,1])];
    assert.equal(inside(point,solidTriangles),true);
    const result=core.evaluate({revision:1,fragments:[{...fragment(parent),solidTriangles},fragment(child)]});
    assert.deepEqual(result.unrootedIds,['native/child']);assert.equal(result.ports[0].live,false);
  }finally{core.dispose();}
});

test('a common witness on the current material boundary is not a strict interior proof',()=>{
  const {core,parent,child,fragment}=fixture('native');try{
    const solidTriangles=boxTriangles([0,0,0],[2,.5,1]);
    const result=core.evaluate({revision:1,fragments:[{...fragment(parent),solidTriangles},fragment(child)]});
    assert.deepEqual(result.unrootedIds,['native/child']);assert.equal(result.ports[0].live,false);
    assert.ok(result.unresolvedBindings.some(b=>b.reason==='strict-embed-witness-not-in-surviving-interior'));
  }finally{core.dispose();}
});

test('an explicitly supplied open solid cannot fall back to the original closed parent',()=>{
  const {core,parent,child,fragment}=fixture();try{
    const result=core.evaluate({revision:1,fragments:[{...fragment(parent),solidTriangles:bottom(parent,0).map(i=>parent.triangles[i])},fragment(child)]});
    assert.deepEqual(result.unrootedIds,['cold/child']);assert.equal(result.ports[0].live,false);
    assert.ok(result.unresolvedBindings.some(b=>b.reason==='strict-embed-effective-solid-unproven'));
  }finally{core.dispose();}
});
