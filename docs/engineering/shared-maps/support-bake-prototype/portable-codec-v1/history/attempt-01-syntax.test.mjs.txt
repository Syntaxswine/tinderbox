function seal(input){
  const b=new Uint8Array(input),v=new DataView(b.buffer);v.setUint32(8,b.length,true);
  let c=0xffffffff;for(let i=0;i<b.length-4;i++){c^=b[i];for(let n=0;n<8;n++)c=(c>>>1)^((c&1)?0xedb88320:0);}
  v.setUint32(b.length-4,(c^0xffffffff)>>>0,true);return b;
}
function change(input,fn){const b=new Uint8Array(input);fn(b,new DataView(b.buffer));return seal(b);}
function uvar(n){const a=[];do{const low=n%128;n=Math.floor(n/128);a.push(low|(n?128:0));}while(n);return a;}
function replace(input,start,end,data){const out=new Uint8Array(input.length-(end-start)+data.length);out.set(input.subarray(0,start));out.set(data,start);out.set(input.subarray(end),start+data.length);return seal(out);}
// Trusted-wire location helper for surgical CRC-valid corruptions, not a decoder.
function locate(b){
  const view=new DataView(b.buffer,b.byteOffset,b.byteLength),dictionary=[],edges=[],objects=[],values=[];
  let at=24;
  function uv(){const start=at;let n=0,p=1;while(true){const x=b[at++];n+=(x&127)*p;if(!(x&128))return {start,end:at,value:n};p*=128;}}
  for(let i=0;i<view.getUint32(12,true);i++){
    const len=uv(),start=at;at+=len.value;dictionary.push({...len,dataStart:start,dataEnd:at,text:new TextDecoder().decode(b.subarray(start,at))});
  }
  const bodyStart=at;
  function walk(path){
    const start=at,tag=b[at++];values.push({path,tag,start});
    if(tag===3){at+=8;return;}if(tag===4){uv();return;}if(tag<3)return;
    if(tag===5){const count=uv();for(let i=0;i<count.value;i++)walk(path+'/'+i);return;}
    if(tag===6){
      const count=uv(),keys=[];objects.push({path,keys});
      for(let i=0;i<count.value;i++){const key=uv();keys.push({...key,text:dictionary[key.value].text});walk(path+'/'+dictionary[key.value].text);}return;
    }
    if(tag===7){const count=uv(),pairs=[];for(let i=0;i<count.value;i++)pairs.push([uv(),uv()]);edges.push({path,count,pairs});return;}
    throw Error('Unexpected trusted test wire tag');
  }
  walk('');assert.equal(at,b.length-4);return {dictionary,edges,objects,values,bodyStart};
}

    maxBytes:wire.length-1,maxEntries:3,maxEdgesPerEntry:11,maxEdges:27,
    maxIds:locations.dictionary.length-1,maxIdBytes:Math.max(...locations.dictionary.map(d=>d.value))-1,
