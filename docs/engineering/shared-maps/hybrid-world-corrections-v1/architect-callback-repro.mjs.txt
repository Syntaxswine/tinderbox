import {sceneCompileDiagnostics} from '../../../workbenches/furnished-house/scene.js';

const f=world();

  plan=f.activation.prepare({whole:['floor-owner']});
  f.trial.supportGraph.sourcesAdded=()=>{throw Error('Injected support notification failure');};

  try{plan.publish();}catch(error){failure=error.message;}
  const result={failure,publicationAdvanced:f.activation.state()!==before,
    fuelAdded:f.trial.metrics().fuel.added,coldReserveBefore:stock.coldReserve,
    coldReserveAfter:f.ledger.metrics().coldReserve,wholeClaimed:f.queries.isClaimedPart('floor'),
    physicalRecordsAdded:f.trial.scene.index.surfaces.size-physical,
    compiler:sceneCompileDiagnostics(),trial:trialDiagnostics(),actualNewYorkCalls:0};
  assert.equal(failure,'Injected support notification failure');
  result.partialPublicationObserved=result.fuelAdded>0&&!result.publicationAdvanced&&!result.wholeClaimed&&result.coldReserveAfter===stock.coldReserve;
  assert.equal(result.partialPublicationObserved,true);
  fs.writeFileSync('artifacts/qa/m29-architect-intake/hybrid-callback-repro.json',JSON.stringify(result,null,2)+'\n');
  console.log(JSON.stringify(result));
}finally{plan?.dispose();f.dispose();}
assert.equal(trialDiagnostics().created,trialDiagnostics().disposed);
