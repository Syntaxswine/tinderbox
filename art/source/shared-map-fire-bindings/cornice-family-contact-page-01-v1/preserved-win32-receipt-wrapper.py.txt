"""Owned, finite, hidden source-check children; no service or retained helper."""
import ctypes
import datetime
import json
import os
import shutil
import subprocess
import time
from pathlib import Path
OWNER="01a17ea0-1362-7d13-a42d-f4a5fcff8902"

def identity(p,known=None):
    if os.name!="nt":
        return {"pid":p.pid,"creationFileTime":None,"executable":None,"creationTimeUtc":None}
    class FT(ctypes.Structure):
        _fields_=[("low",ctypes.c_ulong),("high",ctypes.c_ulong)]
    kernel=ctypes.WinDLL("kernel32",use_last_error=True)
    kernel.GetProcessTimes.argtypes=[ctypes.c_void_p,*([ctypes.POINTER(FT)]*4)]
    kernel.QueryFullProcessImageNameW.argtypes=[ctypes.c_void_p,ctypes.c_ulong,ctypes.c_wchar_p,ctypes.POINTER(ctypes.c_ulong)]
    values=[FT() for _ in range(4)]
    if not kernel.GetProcessTimes(int(p._handle),*[ctypes.byref(v) for v in values]):
        raise OSError(ctypes.get_last_error(),"GetProcessTimes failed for owned child")
    ticks=[(v.high<<32)|v.low for v in values]
    n=ctypes.c_ulong(32768);buf=ctypes.create_unicode_buffer(n.value)
    image_source="queried-from-owned-handle"
    if not kernel.QueryFullProcessImageNameW(int(p._handle),0,buf,ctypes.byref(n)):
        if p.poll() is not None and known and str(ticks[0])==known["creationFileTime"] and p.pid==known["pid"]:
            executable=known["executable"];image_source="captured-while-live; same exited handle/creation time"
        else:raise OSError(ctypes.get_last_error(),"QueryFullProcessImageNameW failed for owned child")
    else:executable=buf.value
    created=datetime.datetime(1601,1,1,tzinfo=datetime.timezone.utc)+datetime.timedelta(microseconds=ticks[0]//10)
    return {"pid":p.pid,"creationFileTime":str(ticks[0]),"creationTimeUtc":created.isoformat(),
            "executable":executable,"executableEvidence":image_source,"kernelCpuMs":ticks[2]/10000,"userCpuMs":ticks[3]/10000}

def journal(path,event,value):
    if path:
        with Path(path).open("a",encoding="utf-8",newline="\n") as f:
            f.write(json.dumps({"event":event,**value},separators=(",",":"))+"\n")
            f.flush();os.fsync(f.fileno())

def run_owned(command,cwd,purpose,regions,timeout=60,input_bytes=None,journal_path=None):
    assert 0<regions<=256 and 0<timeout<=60
    started=datetime.datetime.now(datetime.timezone.utc);clock=time.perf_counter()
    flags=getattr(subprocess,"CREATE_NO_WINDOW",0)
    p=subprocess.Popen(command,cwd=cwd,stdin=subprocess.PIPE if input_bytes is not None else subprocess.DEVNULL,
        stdout=subprocess.PIPE,stderr=subprocess.PIPE,creationflags=flags)
    initial=identity(p)
    receipt={**initial,"ownerTask":OWNER,"project":"Tinderbox","purpose":purpose,"regions":regions,
        "cwd":str(Path(cwd).resolve()),"arguments":command[1:],"startedAt":started.isoformat(),
        "deadlineSeconds":timeout,"intendedExpiry":(started+datetime.timedelta(seconds=timeout)).isoformat(),
        "port":None,"retained":False,"endCondition":"finite source page completes or its deadline expires",
        "stopMethod":"exact owned Popen process handle; verify captured PID/creation time before timeout termination"}
    journal(journal_path,"started",receipt)
    timed_out=False
    try:
        try:
            stdout,stderr=p.communicate(input_bytes,timeout=timeout)
        except subprocess.TimeoutExpired:
            timed_out=True
            current=identity(p)
            assert current["pid"]==initial["pid"] and current["creationFileTime"]==initial["creationFileTime"]
            assert current["executable"]==initial["executable"]
            p.kill();stdout,stderr=p.communicate(timeout=5)
    finally:
        if p.poll() is None:
            current=identity(p)
            assert current["pid"]==initial["pid"] and current["creationFileTime"]==initial["creationFileTime"]
            p.kill();p.wait(timeout=5)
    final=identity(p,initial)
    receipt.update({"exitCode":p.returncode,"timedOut":timed_out,"exited":p.poll() is not None,
        "endedAt":datetime.datetime.now(datetime.timezone.utc).isoformat(),
        "elapsedMs":(time.perf_counter()-clock)*1000,
        "kernelCpuMs":final.get("kernelCpuMs"),"userCpuMs":final.get("userCpuMs"),
        "releasedPorts":[],"postExitExecutableEvidence":final.get("executableEvidence"),"cleanup":"owned finite child exited; no port was opened"})
    journal(journal_path,"exited",receipt)
    return stdout,stderr,receipt
